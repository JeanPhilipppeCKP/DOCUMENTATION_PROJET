    📐 Inventaire des Mesures¶

    Toutes les mesures du modèle, classées par table, avec leur contexte de calcul détecté.

    Vue d'ensemble des contextes de filtre¶

    Fonction DAX Rôle en langage métier
    CALCULATE Recalcule une mesure en modifiant le contexte (filtres actifs)
    FILTER Applique un filtre supplémentaire sur une table
    ALL Ignore tous les filtres sur une table ou colonne
    ALLEXCEPT Ignore tous les filtres sauf ceux spécifiés
    ALLSELECTED Conserve uniquement les filtres de l'utilisateur (visuels)
    KEEPFILTERS Conserve les filtres existants en plus du nouveau
    USERELATIONSHIP Active une relation inactive pour ce calcul
    TREATAS Simule une relation pour propager un filtre
    Simple Agrégation directe sans manipulation de contexte

    _u_cas_abandon¶

    CALCULATE(
    [U-CAS],filter(FACT_OPP,
    'FACT_OPP'[statut_opportunité] IN { "ABANDONNE" })
)

    _u_cas_ec_qualifié¶

    CALCULATE(
    [U-CAS], FILTER(FACT_OPP,
    'FACT_OPP'[stage]
        IN { "4 Negotiation", "5 Handover", "3 Proposal" })
)

    U-CAS_N-1¶

    CALCULATE([_u_cas_signé], SAMEPERIODLASTYEAR(DIM_CALENDAR[Date]))

    Win_ratio_CA¶

    DIVIDE(CALCULATE(
    [U-CAS],
    'FACT_OPP'[statut_opportunité] IN { "GAGNE" })
,[Total_CAS_S/L/A])

    U-TCV MONTHLY YTD¶

    var max_extract_date =max('FACT_OPP_MONTHLY'[extract_date])

var june = CONVERT(year(max_extract_date)&"06",INTEGER)

var htd = calculate(sum('FACT_OPP_MONTHLY'[product_tcv_converted]), 'FACT_OPP_MONTHLY'[report_month]=june)

return if(month(max_extract_date)>6, sum('FACT_OPP_MONTHLY'[product_tcv_converted]) + htd, sum('FACT_OPP_MONTHLY'[product_tcv_converted]))

    Win Ratio Global Context¶

    var won = calculate(SUM('Win Ratio Context'[CLOSED WON]),'Win Ratio Context'[DATE START]<> BLANK())
var all_closed = calculate(sum('Win Ratio Context'[ALL CLOSED]),'Win Ratio Context'[DATE START]<> BLANK())

return DIVIDE(won,all_closed)
